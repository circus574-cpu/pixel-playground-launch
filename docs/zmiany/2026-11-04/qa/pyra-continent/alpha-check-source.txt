import sharp from 'sharp';import fs from 'node:fs/promises';
const root='art/production/pyra-continent-v2/',m=JSON.parse(await fs.readFile(root+'manifest.json','utf8'));const result=[],tiles=[];
for (let i=0;i<m.assets.length;i++) {const a=m.assets[i],{data,info}=await sharp(root+a.url).ensureAlpha().raw().toBuffer({resolveWithObject:true});let edges=0,opaque=0;for(let y=0;y<info.height;y++)for(let x=0;x<info.width;x++){const alpha=data[(y*info.width+x)*4+3];if(alpha>0)opaque++;if((!x||!y||x===info.width-1||y===info.height-1)&&alpha>0)edges++;}result.push({id:a.id,width:info.width,height:info.height,edgeNonzeroAlpha:edges,coverage:opaque/(info.width*info.height)});if(!a.id.includes('fx-')){const tile=await sharp(root+a.url).resize(145,125,{fit:'contain',background:'#30343b'}).flatten({background:'#30343b'}).png().toBuffer();tiles.push({input:tile,left:(i%8)*150,top:Math.floor(i/8)*130});}}
await fs.writeFile('docs/zmiany/2026-11-04/qa/pyra-continent/alpha-check.json',JSON.stringify(result,null,2));await sharp({create:{width:1200,height:910,channels:3,background:'#30343b'}}).composite(tiles).jpeg({quality:85}).toFile('/tmp/pyra-runtime-contact.jpg');console.log('52 assets inspected');
