import { chromium } from '/workspace/ki-combat/node_modules/playwright/index.mjs';
const browser=await chromium.launch({executablePath:'/usr/bin/chromium',headless:true,args:['--no-sandbox','--disable-webgl']});
try {
 const page=await browser.newPage({viewport:{width:1280,height:720}});const errors=[];page.on('pageerror',e=>errors.push(e.message));
 const response=await page.goto('https://kilegends.pl/',{waitUntil:'domcontentloaded',timeout:60000});
 if(response.status()!==200)throw Error('Unexpected live HTTP '+response.status());
 await page.getByRole('button',{name:'Wybierz: Planetarni',exact:true}).click({timeout:60000});
 await page.getByRole('button',{name:'Rozpocznij przygodę',exact:true}).click();
 await page.getByRole('button',{name:'Pokaż statystyki postaci',exact:true}).click({timeout:90000});
 const dialog=page.getByRole('dialog',{name:'Statystyki postaci',exact:true});await dialog.waitFor({state:'visible'});
 for(const label of ['Technika','Unik (%)','Premia szybkości ataku (%)'])if(await dialog.getByText(label,{exact:true}).count()!==1)throw Error('Missing live stat '+label);
 await page.screenshot({path:'/tmp/ki-live-merged-statistics.png'});
 await dialog.getByRole('button',{name:'Zamknij okno',exact:true}).click();
 if(errors.length)throw Error(JSON.stringify(errors));
 console.log('PASS live HTTP200, race selection, world entry, character statistics and close; zero page errors');
} finally {await browser.close();}
